def needs(kind,values):
    if kind in NEEDS:
        return NEEDS[kind]
    known=frozenset(values).intersection((A,B,P))
    pair={frozenset((A,B)):'ab',frozenset((A,P)):'ap',frozenset((B,P)):'bp',frozenset((A,B,P)):'three_known'}[known]
    more=['unknown'] if kind in ('T6','T9') else []
    return [['rank',*more],[pair,*more]]

def sample():
    cfg=read(mini.CONFIG)
    result=[]
    for task, families in zip(TASKS,(REG,TAG)):
        old={}
        for cid,c,y in select_cases(cfg,task):
            if task=='reservation':
                now,book,start=(datetime.fromisoformat(c[k]) for k in ('now','booked_at','starts_at'))
                args=(int((now-book).total_seconds()),int((start-now).total_seconds()))
            else:
                args=tuple(c['initial_state']['args'][0])
            old[args]=(cid,y)
        offset=len(result)
        for kind,inputs in families.items():
            for values in inputs:
                if task=='reservation':
                    age,remaining=values
                    expected=remaining>=86400 or (0<remaining and 0<=age<=600)
                else:
                    expected=next((k for k in (P,B,A) if k in values),'general')
                orig=old.get(values)
                if orig and not (type(orig[1]) is type(expected) and orig[1]==expected):
                    raise ValueError('기존 기준 정답과 불일치')
                result.append(dict(id=('R' if task=='reservation' else 'T')+f'{len(result)-offset+1:02}',task=task,
                    kind=kind,type=KINDS[kind],inputs=list(values),expected=expected,old_id=orig[0] if orig else '',
                    owner_paths=needs(kind,values)))
        subset=result[offset:]
        assert len(subset)==len({tuple(c['inputs']) for c in subset})==50
        assert {c['old_id'] for c in subset if c['old_id']}=={v[0] for v in old.values()}
    return result
